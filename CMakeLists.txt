cmake_minimum_required(VERSION 3.16)
project(franka_gamepad_teleop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Franka 0.20 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(SDL2 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(gamepad_core src/core.cpp src/input.cpp)
target_include_directories(gamepad_core PUBLIC include)
target_link_libraries(gamepad_core PUBLIC Eigen3::Eigen SDL2::SDL2 yaml-cpp Threads::Threads)
target_compile_options(gamepad_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(gamepad_logging src/logging.cpp)
target_link_libraries(gamepad_logging PUBLIC gamepad_core)
target_compile_options(gamepad_logging PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gamepad_teleop src/main.cpp src/gripper.cpp)
target_link_libraries(gamepad_teleop PRIVATE gamepad_core gamepad_logging Franka::Franka)
target_compile_options(gamepad_teleop PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS gamepad_teleop DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  add_executable(gamepad_core_test test/core_test.cpp)
  target_link_libraries(gamepad_core_test PRIVATE gamepad_core)
  add_test(NAME gamepad_core COMMAND gamepad_core_test ${CMAKE_CURRENT_SOURCE_DIR}/config/gamepad.yaml)
  add_executable(gamepad_gripper_test test/gripper_test.cpp src/gripper.cpp)
  # Compile the actual worker against an explicitly fake SDK, without libfranka.
  target_include_directories(gamepad_gripper_test BEFORE PRIVATE test/fake_sdk)
  target_link_libraries(gamepad_gripper_test PRIVATE gamepad_core)
  add_test(NAME gamepad_gripper COMMAND gamepad_gripper_test ${CMAKE_CURRENT_SOURCE_DIR}/config/gamepad.yaml)
  add_executable(gamepad_logging_test test/logging_test.cpp)
  target_link_libraries(gamepad_logging_test PRIVATE gamepad_logging)
  add_test(NAME gamepad_logging COMMAND gamepad_logging_test)
endif()
ament_package()
